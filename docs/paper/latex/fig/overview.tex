\centering
\resizebox{\linewidth}{!}{%
\begin{tikzpicture}[
  font=\sffamily\small,
  tok/.style={minimum size=3.4mm, inner sep=0pt, draw=black!25, fill=black!5},
  keep/.style={tok, fill=cvprblue!75, draw=cvprblue!90},
  ts/.style={rounded corners=1pt, inner sep=1.3pt, font=\sffamily\scriptsize, fill=orange!25, draw=orange!70},
  tsoff/.style={ts, fill=black!4, draw=black!20, text=black!30},
  panel/.style={rounded corners=4pt, draw=black!35, inner sep=6pt},
  box/.style={rounded corners=3pt, draw=black!45, align=center, inner sep=4pt},
  arr/.style={-{Stealth[length=2.4mm]}, thick, black!60},
  title/.style={font=\sffamily\footnotesize\bfseries, anchor=west},
  note/.style={font=\sffamily\scriptsize, anchor=west, align=left}
]
%% ---------------------------------------------------------------- (a) QA pruner
\node[title] at (0, 4.15) {(a) QA pruner: keeps salient / unique tokens};
\foreach \f/\t in {0/0s,1/8s,2/16s,3/24s,4/32s,5/40s} {
  \pgfmathsetmacro{\x}{\f*1.3}
  \ifnum\f=1 \node[tsoff] at (\x+0.36, 3.7) {\t};
  \else\ifnum\f=4 \node[tsoff] at (\x+0.36, 3.7) {\t};
  \else \node[ts] at (\x+0.36, 3.7) {\t}; \fi\fi
  \foreach \i in {0,1,2} \foreach \j in {0,1,2} \node[tok] at (\x+\i*0.36, 3.25-\j*0.36) {};
}
\foreach \f/\i/\j in {0/0/0,0/1/1,2/0/0,2/1/0,2/2/1,2/1/2,3/0/1,3/2/2,5/2/0,5/1/1,5/0/2,5/2/2} {
  \pgfmathsetmacro{\x}{\f*1.3}
  \node[keep] at (\x+\i*0.36, 3.25-\j*0.36) {};
}
\node[note, text=red!70!black, anchor=north west] at (0, 2.4) {8\,s and 32\,s lose every token: their timestamps are gone};

%% ---------------------------------------------------------------- (b) Coverage
\node[title] at (0, 1.55) {(b) Coverage: equal quota per frame, filled by the question};
\foreach \f/\t in {0/0s,1/8s,2/16s,3/24s,4/32s,5/40s} {
  \pgfmathsetmacro{\x}{\f*1.3}
  \node[ts] at (\x+0.36, 1.1) {\t};
  \foreach \i in {0,1,2} \foreach \j in {0,1,2} \node[tok] at (\x+\i*0.36, 0.65-\j*0.36) {};
}
\foreach \f/\i/\j/\k/\l in {0/1/0/2/2,1/0/1/1/2,2/2/0/1/1,3/0/0/0/2,4/1/1/2/0,5/2/2/0/1} {
  \pgfmathsetmacro{\x}{\f*1.3}
  \node[keep] at (\x+\i*0.36, 0.65-\j*0.36) {};
  \node[keep] at (\x+\k*0.36, 0.65-\l*0.36) {};
}
\node[note, text=black!65, anchor=north west] at (0, -0.2) {every frame keeps $\lfloor B/F\rfloor$ tokens: every timestamp survives;\\ the question's attention decides which ones};

\node[panel, fit={(-0.1,4.4) (8.0,-1.0)}] (mem) {};

%% ---------------------------------------------------------------- (c) Contrast
\node[title] at (9.0, 4.15) {(c) Contrast: decode time values against the no-video prior};
\node[box, fill=cvprblue!8, minimum width=3.3cm, minimum height=0.95cm] (pm) at (10.8, 3.2)
  {$\log p(y \mid \mathcal{M}, x)$\\[-1pt]{\scriptsize pruned memory}};
\node[box, fill=black!4, minimum width=3.3cm, minimum height=0.95cm] (p0) at (10.8, 1.75)
  {$\log p(y \mid \varnothing, x)$\\[-1pt]{\scriptsize no visual tokens}};
\node[box, fill=orange!12, text width=4.3cm, minimum height=2.4cm] (sc) at (15.6, 2.48)
  {$(1{+}\alpha_t)\log p_{\mathcal{M}}(y) - \alpha_t \log p_{\varnothing}(y)$\\[4pt]
   {\scriptsize applied only to time values\\ $\alpha_t = 1 - \max_y p_{\mathcal{M}}(y)$ (fades when confident)\\
   plausibility: $p_{\mathcal{M}}(y) \geq 0.1 \max p_{\mathcal{M}}$}};
\draw[arr] (pm.east) -- (pm.east -| sc.west);
\draw[arr] (p0.east) -- (p0.east -| sc.west);
\draw[arr] (mem.east |- pm.west) -- node[above, font=\sffamily\scriptsize] {KV} (pm.west);

\node[box, fill=white, font=\ttfamily\scriptsize, align=left, anchor=north west, text width=8.6cm] (out) at (9.0, 0.6)
  {Seen: 12.0 seconds, opens the paper cutter\\
   Seen: 41.0 seconds, puts the cutter away\\
   Answer: about 29 seconds\\
   Time: [12.0, 41.0]};
\node[note, text=black!60, anchor=north west] at (9.0, -0.75) {answer reserve: with 64 tokens left and no \texttt{Answer:} line, write it};
\draw[arr] (sc.south) -- (sc.south |- out.north);
\end{tikzpicture}}
\caption{\textbf{Overview of \method.} \textbf{(a)}~QA-oriented pruners keep salient or unique tokens and can empty
whole frames, removing their timestamps from memory. \textbf{(b)}~Coverage gives every frame (pair) the same share of
each layer's budget and fills it with the tokens the question attends to most, so the whole timeline stays anchored.
\textbf{(c)}~Contrast writes each time value by amplifying the evidence of the pruned memory over the model's
video-free prediction; the strength fades where the model is confident, and the answer reserve guarantees that the
final interval is written.}
\label{fig:overview}
\end{figure*}
